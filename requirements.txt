# This is the SLIM dependency set for the always-on web app (main.py) —
# routes, uploads, status polling, and the live "Ask This Lecture" endpoint.
# It deliberately excludes crewai/pymupdf/pytesseract/pillow/edge-tts: none

# PIPELINE_EXECUTOR=thread dev), see requirements-worker.txt instead.
